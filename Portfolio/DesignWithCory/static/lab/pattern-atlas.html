CR·01

Singleton

One instance, one global access point — and enforced at runtime, not just by the compiler.

✓ strict TS · zero errors ✓ edge cases exercised
01-singleton.ts 
// Singleton — Application Configuration Registry
// Ensures a class has exactly one instance and provides a global access point to it.

class ConfigRegistry {
  private static instance: ConfigRegistry | undefined;
  private readonly settings: Map<string, string>;

  private constructor() {
    if (ConfigRegistry.instance) {
      // TypeScript's `private` keyword only blocks `new` at compile time.
      // This runtime guard closes the gap against bypasses like `new (Cls as any)()`.
      throw new Error("ConfigRegistry is a singleton — use getInstance().");
    }
    this.settings = new Map([["env", "production"]]);
  }

  static getInstance(): ConfigRegistry {
    if (!ConfigRegistry.instance) {
      ConfigRegistry.instance = new ConfigRegistry();
    }
    return ConfigRegistry.instance;
  }

  get(key: string): string | undefined {
    return this.settings.get(key);
  }

  set(key: string, value: string): void {
    this.settings.set(key, value);
  }
}

// --- Demo & edge cases ---
const a = ConfigRegistry.getInstance();
const b = ConfigRegistry.getInstance();
console.log("Same instance:", a === b); // true

a.set("featureFlag", "on");
console.log("b sees a's mutation:", b.get("featureFlag")); // "on"

try {
  // Bypass attempt: cast to `any` to skip the compiler's private-constructor check.
  new (ConfigRegistry as unknown as new () => ConfigRegistry)();
  console.log("Bypass succeeded (unexpected)");
} catch (e) {
  console.log("Bypass blocked at runtime:", (e as Error).message);
}

Factory Method

Defers which concrete class gets built to a subclass, so callers never name it.

✓ strict TS · zero errors ✓ edge cases exercised
02-factory-method.ts 
// Factory Method — Notification Senders
// Lets a base class defer instantiation of the concrete product to subclasses,
// so client code never names the concrete class it's working with.

interface Notifier {
  send(message: string): void;
}

class EmailNotifier implements Notifier {
  constructor(private readonly address: string) {}
  send(message: string): void {
    console.log(`Email to ${this.address}: ${message}`);
  }
}

class SmsNotifier implements Notifier {
  constructor(private readonly phone: string) {
    if (!/^\+?[0-9]{7,15}$/.test(phone)) {
      throw new Error(`Invalid phone number: ${phone}`);
    }
  }
  send(message: string): void {
    console.log(`SMS to ${this.phone}: ${message}`);
  }
}

abstract class NotifierCreator {
  // The factory method — subclasses decide which concrete product to build.
  protected abstract createNotifier(): Notifier;

  // Client-facing template method; it never sees a concrete Notifier class.
  notify(message: string): void {
    const notifier = this.createNotifier();
    notifier.send(message);
  }
}

class EmailNotifierCreator extends NotifierCreator {
  constructor(private readonly address: string) {
    super();
  }
  protected createNotifier(): Notifier {
    return new EmailNotifier(this.address);
  }
}

class SmsNotifierCreator extends NotifierCreator {
  constructor(private readonly phone: string) {
    super();
  }
  protected createNotifier(): Notifier {
    return new SmsNotifier(this.phone);
  }
}

// --- Demo & edge cases ---
const creators: NotifierCreator[] = [
  new EmailNotifierCreator("ops@example.com"),
  new SmsNotifierCreator("+15551234567"),
];
for (const c of creators) c.notify("Deployment finished");

try {
  new SmsNotifierCreator("not-a-phone").notify("test");
} catch (e) {
  console.log("Rejected invalid input:", (e as Error).message);
}

Abstract Factory

Produces whole families of matching objects, guaranteeing they're never mixed.

✓ strict TS · zero errors ✓ edge cases exercised
03-abstract-factory.ts 
// Abstract Factory — UI Widget Kits
// Produces families of related objects (Button + Checkbox) without the client
// naming their concrete classes — guaranteeing the family never gets mixed.

interface Button {
  render(): string;
}
interface Checkbox {
  render(): string;
}

class LightButton implements Button {
  render(): string { return "[ Light Button ]"; }
}
class LightCheckbox implements Checkbox {
  render(): string { return "[x] Light Checkbox"; }
}
class DarkButton implements Button {
  render(): string { return "▓ Dark Button ▓"; }
}
class DarkCheckbox implements Checkbox {
  render(): string { return "[■] Dark Checkbox"; }
}

interface WidgetFactory {
  createButton(): Button;
  createCheckbox(): Checkbox;
}

class LightWidgetFactory implements WidgetFactory {
  createButton(): Button { return new LightButton(); }
  createCheckbox(): Checkbox { return new LightCheckbox(); }
}
class DarkWidgetFactory implements WidgetFactory {
  createButton(): Button { return new DarkButton(); }
  createCheckbox(): Checkbox { return new DarkCheckbox(); }
}

type Theme = "light" | "dark";

function getFactory(theme: Theme): WidgetFactory {
  switch (theme) {
    case "light":
      return new LightWidgetFactory();
    case "dark":
      return new DarkWidgetFactory();
    default: {
      // Exhaustiveness check: a new Theme member breaks the build here, not at runtime.
      const exhaustive: never = theme;
      throw new Error(`Unhandled theme: ${exhaustive}`);
    }
  }
}

// --- Demo & edge cases ---
function renderForm(factory: WidgetFactory): void {
  console.log(factory.createButton().render());
  console.log(factory.createCheckbox().render());
}

renderForm(getFactory("light"));
renderForm(getFactory("dark"));

// Every product pulled from one factory belongs to the same family —
// there is no code path that can hand back a LightButton with a DarkCheckbox.
const darkFactory = getFactory("dark");
const consistent =
  darkFactory.createButton() instanceof DarkButton &&
  darkFactory.createCheckbox() instanceof DarkCheckbox;
console.log("Family stays consistent:", consistent);
CR·04

Builder

Assembles a complex, immutable object through validated, chainable steps.

✓ strict TS · zero errors ✓ edge cases exercised
04-builder.ts 
// Builder — HTTP Request Builder
// Assembles a complex, immutable object step by step, validating as it goes
// instead of exposing a giant constructor with many optional parameters.

interface HttpRequest {
  readonly url: string;
  readonly method: "GET" | "POST" | "PUT" | "DELETE";
  readonly headers: Readonly<Record<string, string>>;
  readonly body: string | undefined;
}

class HttpRequestBuilder {
  private url?: string;
  private method: HttpRequest["method"] = "GET";
  private headers: Record<string, string> = {};
  private body?: string;

  setUrl(url: string): this {
    if (!/^https?:\/\//.test(url)) {
      throw new Error(`URL must start with http(s)://, got "${url}"`);
    }
    this.url = url;
    return this;
  }

  setMethod(method: HttpRequest["method"]): this {
    this.method = method;
    return this;
  }

  addHeader(key: string, value: string): this {
    this.headers = { ...this.headers, [key]: value };
    return this;
  }

  setBody(body: string): this {
    if (this.method === "GET") {
      throw new Error("GET requests cannot have a body");
    }
    this.body = body;
    return this;
  }

  build(): HttpRequest {
    if (!this.url) {
      throw new Error("Cannot build a request without a URL");
    }
    return Object.freeze({
      url: this.url,
      method: this.method,
      headers: Object.freeze({ ...this.headers }),
      body: this.body,
    });
  }
}

// --- Demo & edge cases ---
const request = new HttpRequestBuilder()
  .setUrl("https://api.example.com/orders")
  .setMethod("POST")
  .addHeader("Content-Type", "application/json")
  .setBody(JSON.stringify({ item: "widget" }))
  .build();

console.log(request);

try {
  new HttpRequestBuilder().build(); // missing required URL
} catch (e) {
  console.log("Rejected:", (e as Error).message);
}

try {
  new HttpRequestBuilder().setUrl("https://x.com").setBody("oops"); // GET + body
} catch (e) {
  console.log("Rejected:", (e as Error).message);
}

// Two builds from the same builder don't alias each other's headers object.
const builder = new HttpRequestBuilder().setUrl("https://x.com/a");
const first = builder.build();
builder.addHeader("X-Trace", "abc");
const second = builder.build();
console.log("First build unaffected by later mutation:", first.headers, second.headers);
CR·05

Prototype

Clones an existing instance instead of building from scratch — deep-copying mutable state.

✓ strict TS · zero errors ✓ edge cases exercised
05-prototype.ts 
// Prototype — Cloneable Shapes
// Creates new objects by copying an existing instance instead of building
// from scratch, while avoiding the classic pitfall of aliasing mutable fields.

interface Prototype<T> {
  clone(): T;
}

class Shape implements Prototype<Shape> {
  constructor(
    public x: number,
    public y: number,
    public fillColor: string,
    public tags: string[] = [],
  ) {}

  clone(): Shape {
    // Deep-copy the mutable array so the clone doesn't alias the original's.
    return new Shape(this.x, this.y, this.fillColor, [...this.tags]);
  }
}

class Circle extends Shape {
  constructor(
    x: number,
    y: number,
    fillColor: string,
    public radius: number,
    tags: string[] = [],
  ) {
    super(x, y, fillColor, tags);
  }

  override clone(): Circle {
    // Covariant override — cloning a Circle returns a Circle, not a bare Shape.
    return new Circle(this.x, this.y, this.fillColor, this.radius, [...this.tags]);
  }
}

// --- Demo & edge cases ---
const original = new Circle(10, 10, "#f00", 5, ["logo"]);
const copy = original.clone();
copy.tags.push("duplicate");
copy.x = 99;

console.log("original unaffected by clone's array mutation:", original.tags);
console.log("original unaffected by clone's field mutation:", original.x);
console.log("copy is independent:", copy.tags, copy.x);
console.log("clone preserves the concrete subclass:", copy instanceof Circle);

// Cloning an empty-tags shape must not throw or share a frozen/undefined array.
const bare = new Shape(0, 0, "#000");
const bareClone = bare.clone();
bareClone.tags.push("new-tag");
console.log("original with no initial tags stays empty:", bare.tags.length === 0);

Adapter

✓ strict TS · zero errors ✓ edge cases exercised
06-adapter.ts 
// Adapter — Legacy Weather Client
// Lets a class with an incompatible interface (legacy, Fahrenheit, XML-shaped)
// work behind the interface modern client code actually expects.

interface WeatherService {
  getTemperatureCelsius(city: string): number;
}

// Existing third-party class we cannot modify.
class LegacyWeatherXmlClient {
  fetchXmlReport(city: string): string {
    const fakeFahrenheit = city === "Phoenix" ? 104 : 68;
    return `<report><city>${city}</city><tempF>${fakeFahrenheit}</tempF></report>`;
  }
}

class LegacyWeatherAdapter implements WeatherService {
  constructor(private readonly legacyClient: LegacyWeatherXmlClient) {}

  getTemperatureCelsius(city: string): number {
    const xml = this.legacyClient.fetchXmlReport(city);
    const match = /<tempF>(-?\d+(\.\d+)?)<\/tempF>/.exec(xml);
    if (!match) {
      throw new Error(`Malformed legacy response for "${city}": ${xml}`);
    }
    const fahrenheit = Number(match[1]);
    return Math.round(((fahrenheit - 32) * 5) / 9 * 10) / 10;
  }
}

// --- Demo & edge cases ---
const legacyClient = new LegacyWeatherXmlClient();
const service: WeatherService = new LegacyWeatherAdapter(legacyClient);
console.log("Phoenix °C:", service.getTemperatureCelsius("Phoenix"));
console.log("Boston °C:", service.getTemperatureCelsius("Boston"));

try {
  const brokenClient = { fetchXmlReport: () => "<report><city>Nowhere</city></report>" };
  const badAdapter = new LegacyWeatherAdapter(brokenClient as LegacyWeatherXmlClient);
  badAdapter.getTemperatureCelsius("Nowhere");
} catch (e) {
  console.log("Rejected malformed legacy response:", (e as Error).message);
}
ST·02

Bridge

Separates an abstraction from its implementation so each can evolve on its own.

✓ strict TS · zero errors ✓ edge cases exercised
07-bridge.ts 
// Bridge — Remote Controls & Devices
// Splits an "abstraction" hierarchy (remotes) from an "implementation"
// hierarchy (devices) so each can grow independently of the other.

interface Device {
  isEnabled(): boolean;
  enable(): void;
  disable(): void;
  getVolume(): number;
  setVolume(percent: number): void;
}

class Television implements Device {
  private on = false;
  private volume = 30;
  isEnabled(): boolean { return this.on; }
  enable(): void { this.on = true; }
  disable(): void { this.on = false; }
  getVolume(): number { return this.volume; }
  setVolume(percent: number): void {
    this.volume = Math.min(100, Math.max(0, percent));
  }
}

class Radio implements Device {
  private on = false;
  private volume = 50;
  isEnabled(): boolean { return this.on; }
  enable(): void { this.on = true; }
  disable(): void { this.on = false; }
  getVolume(): number { return this.volume; }
  setVolume(percent: number): void {
    this.volume = Math.min(100, Math.max(0, percent));
  }
}

// Abstraction — holds a reference to (bridges to) a Device implementation.
class RemoteControl {
  constructor(protected device: Device) {}

  togglePower(): void {
    this.device.isEnabled() ? this.device.disable() : this.device.enable();
  }
  volumeDown(): void { this.device.setVolume(this.device.getVolume() - 10); }
  volumeUp(): void { this.device.setVolume(this.device.getVolume() + 10); }
}

// Refined abstraction — extends behavior without touching any Device class.
class AdvancedRemoteControl extends RemoteControl {
  mute(): void { this.device.setVolume(0); }
}

// --- Demo & edge cases ---
const tv = new Television();
const remote = new AdvancedRemoteControl(tv);
remote.togglePower();
remote.volumeUp();
remote.volumeUp();
console.log("TV volume:", tv.getVolume(), "on:", tv.isEnabled());
remote.mute();
console.log("TV muted volume:", tv.getVolume());

const radio = new Radio();
const basicRemote = new RemoteControl(radio);
for (let i = 0; i < 6; i++) basicRemote.volumeDown(); // would go negative unclamped
console.log("Radio volume clamps at 0, not negative:", radio.getVolume());

for (let i = 0; i < 12; i++) basicRemote.volumeUp(); // would exceed 100 unclamped
console.log("Radio volume clamps at 100, not over:", radio.getVolume());

Composite

Arranges objects into trees so a single leaf and a whole branch share one interface.

✓ strict TS · zero errors ✓ edge cases exercised
08-composite.ts 
// Composite — Filesystem Tree
// Composes files and directories into a tree and lets client code treat a
// single file and an entire directory through the exact same interface.

interface FileSystemNode {
  readonly name: string;
  getSize(): number;
  print(indent?: string): string;
}

class FileLeaf implements FileSystemNode {
  constructor(public readonly name: string, private readonly sizeBytes: number) {
    if (sizeBytes < 0) throw new Error(`File size cannot be negative: ${sizeBytes}`);
  }
  getSize(): number { return this.sizeBytes; }
  print(indent = ""): string { return `${indent}${this.name} (${this.sizeBytes}B)`; }
}

class Directory implements FileSystemNode {
  private readonly children: FileSystemNode[] = [];
  constructor(public readonly name: string) {}

  add(node: FileSystemNode): this {
    if (node === this) throw new Error("A directory cannot contain itself");
    this.children.push(node);
    return this;
  }

  getSize(): number {
    // Uniform recursion: a child may itself be a Directory or a FileLeaf.
    return this.children.reduce((total, child) => total + child.getSize(), 0);
  }

  print(indent = ""): string {
    const lines = [`${indent}${this.name}/`];
    for (const child of this.children) lines.push(child.print(indent + "  "));
    return lines.join("\n");
  }
}

// --- Demo & edge cases ---
const root = new Directory("project");
const src = new Directory("src");
src.add(new FileLeaf("index.ts", 1200)).add(new FileLeaf("utils.ts", 800));
root.add(src).add(new FileLeaf("README.md", 400));

console.log(root.print());
console.log("Total size (leaves + composites, uniformly):", root.getSize(), "bytes");

try {
  root.add(root); // cyclic containment would infinite-loop getSize()
} catch (e) {
  console.log("Rejected cycle:", (e as Error).message);
}

const empty = new Directory("empty");
console.log("Empty directory sums to 0, not NaN:", empty.getSize());

try {
  new FileLeaf("bad.txt", -1);
} catch (e) {
  console.log("Rejected negative size:", (e as Error).message);
}

Decorator

Adds responsibilities to an object at runtime by wrapping it in layers.

✓ strict TS · zero errors ✓ edge cases exercised
09-decorator.ts 
// Decorator — Beverage Add-ons
// Wraps an object at runtime to layer on new responsibilities (cost,
// description) without altering the wrapped object or subclassing per combo.

interface Beverage {
  cost(): number;
  describe(): string;
}

class Espresso implements Beverage {
  cost(): number { return 2.5; }
  describe(): string { return "Espresso"; }
}

abstract class BeverageDecorator implements Beverage {
  constructor(protected readonly wrapped: Beverage) {}
  abstract cost(): number;
  abstract describe(): string;
}

class MilkDecorator extends BeverageDecorator {
  cost(): number { return this.wrapped.cost() + 0.5; }
  describe(): string { return `${this.wrapped.describe()} + Milk`; }
}

class SugarDecorator extends BeverageDecorator {
  cost(): number { return this.wrapped.cost() + 0.2; }
  describe(): string { return `${this.wrapped.describe()} + Sugar`; }
}

class WhippedCreamDecorator extends BeverageDecorator {
  cost(): number { return Math.round((this.wrapped.cost() + 0.7) * 100) / 100; }
  describe(): string { return `${this.wrapped.describe()} + Whipped Cream`; }
}

// --- Demo & edge cases ---
let order: Beverage = new Espresso();
order = new MilkDecorator(order);
order = new SugarDecorator(order);
order = new WhippedCreamDecorator(order);
console.log(order.describe(), "→ $" + order.cost().toFixed(2));

// Wrapping order changes the description text but not the final total —
// use an epsilon comparison since these are floating-point sums.
const differentOrder = new SugarDecorator(new MilkDecorator(new Espresso()));
const totalsMatch = Math.abs(order.cost() - 0.7 - differentOrder.cost()) < 1e-9;
console.log("Totals agree regardless of wrap order:", totalsMatch);

// The undecorated base case still works on its own.
const plain: Beverage = new Espresso();
console.log(plain.describe(), "→ $" + plain.cost().toFixed(2));
ST·05

Facade

Hides a tangle of subsystems behind one simple, coordinated entry point.

✓ strict TS · zero errors ✓ edge cases exercised
10-facade.ts 
// Facade — Checkout Facade
// Gives client code one simple entry point in front of several subsystems
// (inventory, payment, shipping), including coordinating their failure paths.

class InventoryService {
  private stock = new Map<string, number>([["sku-1", 5]]);
  reserve(sku: string, qty: number): void {
    const available = this.stock.get(sku) ?? 0;
    if (available < qty) throw new Error(`Insufficient stock for ${sku}`);
    this.stock.set(sku, available - qty);
  }
  release(sku: string, qty: number): void {
    this.stock.set(sku, (this.stock.get(sku) ?? 0) + qty);
  }
}

class PaymentService {
  charge(amountCents: number): string {
    if (amountCents <= 0) throw new Error("Charge amount must be positive");
    if (amountCents > 100_000) throw new Error("Card declined: over limit");
    return `txn_${amountCents}_${Date.now()}`;
  }
}

class ShippingService {
  schedule(sku: string, qty: number): string {
    return `ship_${sku}_x${qty}`;
  }
}

// Facade — hides three subsystems behind one call and rolls back on failure.
class CheckoutFacade {
  private readonly inventory = new InventoryService();
  private readonly payment = new PaymentService();
  private readonly shipping = new ShippingService();

  placeOrder(
    sku: string,
    qty: number,
    amountCents: number,
  ): { txnId: string; shipmentId: string } {
    this.inventory.reserve(sku, qty);
    try {
      const txnId = this.payment.charge(amountCents);
      const shipmentId = this.shipping.schedule(sku, qty);
      return { txnId, shipmentId };
    } catch (err) {
      // A failed payment must not strand reserved stock — roll it back.
      this.inventory.release(sku, qty);
      throw err;
    }
  }
}

// --- Demo & edge cases ---
const checkout = new CheckoutFacade();
console.log("Order placed:", checkout.placeOrder("sku-1", 2, 4999));

try {
  checkout.placeOrder("sku-1", 10, 1000); // only 3 left in stock
} catch (e) {
  console.log("Rejected:", (e as Error).message);
}

try {
  checkout.placeOrder("sku-1", 1, 999_999); // stock reserved, then payment fails
} catch (e) {
  console.log("Rejected (stock rolled back):", (e as Error).message);
}
console.log("Stock recovered:", checkout.placeOrder("sku-1", 3, 1500));
ST·06

Flyweight

Shares heavy intrinsic state across many lightweight instances to save memory.

✓ strict TS · zero errors ✓ edge cases exercised
11-flyweight.ts 
// Flyweight — Forest Renderer
// Shares heavy, repeated intrinsic state (species/texture) across thousands
// of instances, so only per-instance extrinsic state (x, y) is duplicated.

interface TreeTypeProps {
  readonly species: string;
  readonly color: string;
  readonly textureId: string;
}

// Flyweight — imagine `textureId` backs a large shared texture buffer.
class TreeType {
  constructor(
    public readonly species: string,
    public readonly color: string,
    public readonly textureId: string,
  ) {}

  draw(x: number, y: number): string {
    return `Drawing ${this.species} (${this.color}) at (${x}, ${y}) via ${this.textureId}`;
  }
}

// Factory ensures identical intrinsic state is never allocated twice.
class TreeTypeFactory {
  private static readonly cache = new Map<string, TreeType>();

  static get(props: TreeTypeProps): TreeType {
    const key = `${props.species}|${props.color}|${props.textureId}`;
    let type = TreeTypeFactory.cache.get(key);
    if (!type) {
      type = new TreeType(props.species, props.color, props.textureId);
      TreeTypeFactory.cache.set(key, type);
    }
    return type;
  }

  static get poolSize(): number {
    return TreeTypeFactory.cache.size;
  }
}

// Context — cheap, holds only extrinsic (per-instance) state.
class Tree {
  constructor(
    private readonly x: number,
    private readonly y: number,
    private readonly type: TreeType,
  ) {}
  draw(): string { return this.type.draw(this.x, this.y); }
}

class Forest {
  private readonly trees: Tree[] = [];

  plant(x: number, y: number, props: TreeTypeProps): void {
    const type = TreeTypeFactory.get(props); // shared, not duplicated per-tree
    this.trees.push(new Tree(x, y, type));
  }

  drawAll(): string[] {
    return this.trees.map((tree) => tree.draw());
  }
}

// --- Demo & edge cases ---
const forest = new Forest();
const oak: TreeTypeProps = { species: "Oak", color: "green", textureId: "tex_oak" };
for (let i = 0; i < 10_000; i++) {
  forest.plant(i % 100, Math.floor(i / 100), oak);
}
forest.plant(5, 5, { species: "Pine", color: "dark-green", textureId: "tex_pine" });

console.log("Trees planted:", forest.drawAll().length); // 10001
console.log("Distinct shared TreeType objects:", TreeTypeFactory.poolSize); // 2

Proxy

Stands in for a real object to gate access to it and defer expensive work.

✓ strict TS · zero errors ✓ edge cases exercised
12-proxy.ts 
// Proxy — Access-Controlled, Lazily-Loaded Document
// Stands in for an expensive real object, enforcing access checks and
// deferring construction of the real object until it's actually needed.

interface Document {
  read(): string;
}

class RealDocument implements Document {
  private readonly content: string;
  constructor(path: string) {
    console.log(`(loading "${path}" from disk...)`); // simulate expensive I/O
    this.content = `Contents of ${path}`;
  }
  read(): string { return this.content; }
}

type Role = "admin" | "editor" | "viewer";

class DocumentProxy implements Document {
  private real?: RealDocument;

  constructor(
    private readonly path: string,
    private readonly requiredRole: Role,
    private readonly userRole: Role,
  ) {}

  read(): string {
    if (!this.hasAccess()) {
      throw new Error(
        `Access denied: "${this.path}" needs role "${this.requiredRole}", got "${this.userRole}"`,
      );
    }
    // Lazily construct the expensive real object only once access is confirmed.
    if (!this.real) {
      this.real = new RealDocument(this.path);
    }
    return this.real.read();
  }

  private hasAccess(): boolean {
    const rank: Record<Role, number> = { viewer: 0, editor: 1, admin: 2 };
    return rank[this.userRole] >= rank[this.requiredRole];
  }
}

// --- Demo & edge cases ---
const asAdmin = new DocumentProxy("payroll.csv", "admin", "admin");
console.log(asAdmin.read()); // triggers the lazy load, then reads

Observer

Lets one subject broadcast state changes to any number of independent listeners.

✓ strict TS · zero errors ✓ edge cases exercised
13-observer.ts 
// Observer — Stock Price Ticker
// Lets a subject broadcast state changes to many independent observers
// without knowing their concrete types, and supports clean unsubscription.

interface Observer<T> {
  update(data: T): void;
}

class Subject<T> {
  private readonly observers = new Set<Observer<T>>();

  subscribe(observer: Observer<T>): () => void {
    this.observers.add(observer);
    return () => this.observers.delete(observer); // unsubscribe handle
  }

  notify(data: T): void {
    for (const observer of this.observers) {
      try {
        observer.update(data);
      } catch (err) {
        // One misbehaving observer must not prevent the others from being notified.
        console.log("Observer failed:", (err as Error).message);
      }
    }
  }
}

interface StockPrice {
  symbol: string;
  price: number;
}

class PriceLogger implements Observer<StockPrice> {
  update(data: StockPrice): void {
    console.log(`[log] ${data.symbol} → $${data.price.toFixed(2)}`);
  }
}

class PriceAlert implements Observer<StockPrice> {
  constructor(private readonly threshold: number) {}
  update(data: StockPrice): void {
    if (data.price > this.threshold) {
      throw new Error(`${data.symbol} crossed alert threshold of $${this.threshold}`);
    }
  }
}

// --- Demo & edge cases ---
const ticker = new Subject<StockPrice>();
const logger = new PriceLogger();
const unsubscribeLogger = ticker.subscribe(logger);
ticker.subscribe(new PriceAlert(100));

ticker.notify({ symbol: "ACME", price: 87.5 });
ticker.notify({ symbol: "ACME", price: 104.2 }); // alert throws; logger still runs

unsubscribeLogger();
ticker.notify({ symbol: "ACME", price: 50 }); // logger no longer prints, no error either
console.log("Notify with zero remaining observers is a safe no-op");

Strategy

Swaps interchangeable algorithms into a context object at runtime.

✓ strict TS · zero errors ✓ edge cases exercised
14-strategy.ts 
// Strategy — Cart Pricing Strategies
// Lets an interchangeable algorithm (how a discount is computed) be swapped
// at runtime without changing the code that uses it.

interface PricingStrategy {
  computeTotal(subtotalCents: number): number;
}

class NoDiscount implements PricingStrategy {
  computeTotal(subtotalCents: number): number { return subtotalCents; }
}

class PercentageDiscount implements PricingStrategy {
  private readonly percent: number;
  constructor(percent: number) {
    this.percent = Math.min(100, Math.max(0, percent)); // clamp bad input
  }
  computeTotal(subtotalCents: number): number {
    return Math.round(subtotalCents * (1 - this.percent / 100));
  }
}

class FlatAmountDiscount implements PricingStrategy {
  constructor(private readonly amountOffCents: number) {}
  computeTotal(subtotalCents: number): number {
    return Math.max(0, subtotalCents - this.amountOffCents); // never negative
  }
}

class ShoppingCart {
  private strategy: PricingStrategy = new NoDiscount();
  constructor(private readonly itemPricesCents: number[]) {}

  setStrategy(strategy: PricingStrategy): void {
    this.strategy = strategy;
  }

  checkout(): number {
    const subtotal = this.itemPricesCents.reduce((sum, price) => sum + price, 0);
    return this.strategy.computeTotal(subtotal);
  }
}

// --- Demo & edge cases ---
const cart = new ShoppingCart([1999, 500, 799]); // $19.99 + $5.00 + $7.99

console.log("No discount:", cart.checkout());

cart.setStrategy(new PercentageDiscount(20));
console.log("20% off:", cart.checkout());

cart.setStrategy(new PercentageDiscount(150)); // over 100% — must clamp, not negative
console.log("Clamped 150% off:", cart.checkout());

cart.setStrategy(new FlatAmountDiscount(999_999)); // far more than subtotal
console.log("Oversized flat discount floors at 0:", cart.checkout());

const emptyCart = new ShoppingCart([]);
emptyCart.setStrategy(new PercentageDiscount(50));
console.log("Empty cart totals to 0, not NaN:", emptyCart.checkout());

Command

Turns a request into an object, enabling queues, logs, and undo/redo.

✓ strict TS · zero errors ✓ edge cases exercised
15-command.ts 
// Command — Undoable Text Editor Actions
// Turns each request into a standalone object so it can be queued, logged,
// and reversed — powering a generic undo/redo history.

interface Command {
  execute(): void;
  undo(): void;
}

class TextDocument {
  private content = "";
  insertAt(index: number, text: string): void {
    this.content = this.content.slice(0, index) + text + this.content.slice(index);
  }
  deleteAt(index: number, length: number): string {
    const removed = this.content.slice(index, index + length);
    this.content = this.content.slice(0, index) + this.content.slice(index + length);
    return removed;
  }
  get text(): string { return this.content; }
}

class InsertCommand implements Command {
  constructor(
    private readonly doc: TextDocument,
    private readonly index: number,
    private readonly text: string,
  ) {}
  execute(): void { this.doc.insertAt(this.index, this.text); }
  undo(): void { this.doc.deleteAt(this.index, this.text.length); }
}

class DeleteCommand implements Command {
  private removedText = "";
  constructor(
    private readonly doc: TextDocument,
    private readonly index: number,
    private readonly length: number,
  ) {}
  execute(): void { this.removedText = this.doc.deleteAt(this.index, this.length); }
  undo(): void { this.doc.insertAt(this.index, this.removedText); }
}

class CommandHistory {
  private readonly undoStack: Command[] = [];
  private readonly redoStack: Command[] = [];

  run(command: Command): void {
    command.execute();
    this.undoStack.push(command);
    this.redoStack.length = 0; // a fresh action invalidates old redo history
  }

  undo(): void {
    const command = this.undoStack.pop();
    if (!command) return; // nothing left to undo — safe no-op
    command.undo();
    this.redoStack.push(command);
  }

  redo(): void {
    const command = this.redoStack.pop();
    if (!command) return; // nothing left to redo — safe no-op
    command.execute();
    this.undoStack.push(command);
  }
}

// --- Demo & edge cases ---
const doc = new TextDocument();
const history = new CommandHistory();

history.run(new InsertCommand(doc, 0, "Hello"));
history.run(new InsertCommand(doc, 5, " World"));
console.log("After inserts:", JSON.stringify(doc.text));

history.undo();
console.log("After one undo:", JSON.stringify(doc.text));

history.redo();
console.log("After redo:", JSON.stringify(doc.text));

history.undo();
history.undo();
history.undo(); // one extra undo beyond available history
console.log("Over-undo is a safe no-op:", JSON.stringify(doc.text));

history.run(new DeleteCommand(doc, 0, 0)); // zero-length delete
console.log("Zero-length delete no-ops safely:", JSON.stringify(doc.text));

State

Lets an object change its behavior by delegating to interchangeable state objects.

✓ strict TS · zero errors ✓ edge cases exercised
16-state.ts 
// State — Document Workflow
// Lets an object change its behavior when its internal state changes, by
// delegating to interchangeable state objects instead of branching on a flag.

interface DocumentState {
  readonly name: string;
  submit(context: DocumentContext): void;
  approve(context: DocumentContext): void;
  reject(context: DocumentContext): void;
}

class DraftState implements DocumentState {
  readonly name = "draft";
  submit(context: DocumentContext): void { context.transitionTo(new ReviewState()); }
  approve(): void { throw new Error("Cannot approve before it has been submitted"); }
  reject(): void { throw new Error("Cannot reject a document that hasn't been submitted"); }
}

class ReviewState implements DocumentState {
  readonly name = "review";
  submit(): void { throw new Error("Document is already under review"); }
  approve(context: DocumentContext): void { context.transitionTo(new PublishedState()); }
  reject(context: DocumentContext): void { context.transitionTo(new DraftState()); }
}

class PublishedState implements DocumentState {
  readonly name = "published";
  submit(): void { throw new Error("Cannot resubmit a published document"); }
  approve(): void { throw new Error("Document is already published"); }
  reject(): void { throw new Error("Cannot reject a document that is already published"); }
}

class DocumentContext {
  private state: DocumentState = new DraftState();

  transitionTo(state: DocumentState): void {
    console.log(`${this.state.name} → ${state.name}`);
    this.state = state;
  }

  submit(): void { this.state.submit(this); }
  approve(): void { this.state.approve(this); }
  reject(): void { this.state.reject(this); }
  get currentState(): string { return this.state.name; }
}

// --- Demo & edge cases ---
const article = new DocumentContext();
article.submit();
article.reject(); // back to draft
article.submit();
article.approve(); // published

try {
  article.approve(); // invalid: already published
} catch (e) {
  console.log("Rejected:", (e as Error).message);
}

try {
  article.submit(); // invalid: published documents can't be resubmitted
} catch (e) {
  console.log("Rejected:", (e as Error).message);
}

console.log("Final state:", article.currentState);

Iterator

Walks a collection's elements through a standard protocol, hiding its internals.

✓ strict TS · zero errors ✓ edge cases exercised
17-iterator.ts 
// Iterator — Circular Buffer
// Walks through a collection's elements via a standard protocol (Symbol.iterator)
// without exposing the buffer's internal array or shift/wrap logic.

class CircularBuffer<T> implements Iterable<T> {
  private readonly items: T[] = [];
  constructor(private readonly capacity: number) {
    if (capacity <= 0) throw new Error("Capacity must be positive");
  }

  push(item: T): void {
    this.items.push(item);
    if (this.items.length > this.capacity) {
      this.items.shift(); // drop the oldest entry once over capacity
    }
  }

  [Symbol.iterator](): Iterator<T> {
    let index = 0;
    const snapshot = [...this.items]; // isolate from mutation during iteration
    return {
      next(): IteratorResult<T> {
        if (index < snapshot.length) {
          return { value: snapshot[index++] as T, done: false };
        }
        return { value: undefined, done: true };
      },
    };
  }
}

// --- Demo & edge cases ---
const buffer = new CircularBuffer<number>(3);
[1, 2, 3, 4, 5].forEach((n) => buffer.push(n));
console.log("Contents (oldest two dropped):", [...buffer]); // [3, 4, 5]

// Two independent iterators over the same collection don't interfere.
const it1 = buffer[Symbol.iterator]();
const it2 = buffer[Symbol.iterator]();
console.log("it1 first:", it1.next().value);
console.log("it2 first (unaffected by it1's position):", it2.next().value);
console.log("it1 second:", it1.next().value);

const empty = new CircularBuffer<number>(1);
console.log("Empty buffer iterates zero times:", [...empty].length === 0);

try {
  new CircularBuffer<number>(0);
} catch (e) {
  console.log("Rejected non-positive capacity:", (e as Error).message);
}
